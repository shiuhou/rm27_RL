  // RM27 isolated numerical diagnostic. Never clip covariance eigenvalues.
  const Eigen::MatrixXd prior = state->_Cov.selfadjointView<Eigen::Upper>();
  Eigen::MatrixXd H_full = Eigen::MatrixXd::Zero(H.rows(), prior.cols());
  for (size_t i = 0; i < H_order.size(); ++i) {
    H_full.block(0, H_order[i]->id(), H.rows(), H_order[i]->size()) +=
        H.block(0, H_id[i], H.rows(), H_order[i]->size());
  }
  const Eigen::MatrixXd noise = R.selfadjointView<Eigen::Upper>();
  const Eigen::MatrixXd innovation = S.selfadjointView<Eigen::Upper>();
  auto dump = [&](const char *reason, const Eigen::MatrixXd &posterior) {
    auto save = [](const char *name, const Eigen::MatrixXd &matrix) {
      std::ofstream out(std::string("/out/covariance_") + name + ".csv");
      out << matrix.format(Eigen::IOFormat(17, Eigen::DontAlignCols, ",", "\n"));
    };
    save("prior", prior); save("H", H_full); save("R", noise);
    save("S", innovation); save("posterior", posterior);
    PRINT_ERROR(RED "RM27_COVARIANCE_FAILURE: %s time=%.17g prior_diag=%.17g posterior_diag=%.17g\n" RESET,
                reason, state->_timestamp, prior.diagonal().minCoeff(), posterior.diagonal().minCoeff());
    std::exit(EXIT_FAILURE);
  };
  Eigen::LLT<Eigen::MatrixXd> factor(innovation);
  if (!prior.allFinite() || !noise.allFinite() || !innovation.allFinite() || factor.info() != Eigen::Success) {
    dump("nonfinite input or failed innovation LLT", prior);
  }
  const char *form = std::getenv("RM27_COVARIANCE_FORM");
  const bool original = form && std::string(form) == "original";
  if (form && !original && std::string(form) != "joseph") dump("unsupported covariance form", prior);
  Eigen::MatrixXd K;
  if (original) {
    // Exact pinned inverse/subtraction algorithm, with failure diagnostics.
    Eigen::MatrixXd Sinv = Eigen::MatrixXd::Identity(R.rows(), R.rows());
    S.selfadjointView<Eigen::Upper>().llt().solveInPlace(Sinv);
    K = M_a * Sinv.selfadjointView<Eigen::Upper>();
    state->_Cov.triangularView<Eigen::Upper>() -= K * M_a.transpose();
    state->_Cov = state->_Cov.selfadjointView<Eigen::Upper>();
  } else {
    // Solve the innovation system without forming its inverse. Joseph form
    // preserves the full cross-covariances and the supplied measurement noise.
    Eigen::SelfAdjointEigenSolver<Eigen::MatrixXd> prior_factor(prior, Eigen::EigenvaluesOnly);
    const double prior_scale = std::max(1., prior.cwiseAbs().maxCoeff());
    const double prior_roundoff = 64. * prior.rows() * std::numeric_limits<double>::epsilon() * prior_scale;
    if (prior_factor.info() != Eigen::Success || prior_factor.eigenvalues().minCoeff() < -prior_roundoff)
      dump("invalid prior covariance", prior);
    K = factor.solve(M_a.transpose()).transpose();
    const Eigen::MatrixXd residual = Eigen::MatrixXd::Identity(prior.rows(), prior.cols()) - K * H_full;
    const Eigen::MatrixXd posterior = residual * prior * residual.transpose() + K * noise * K.transpose();
    state->_Cov = (0.5 * (posterior + posterior.transpose())).eval();
  }
  if (!K.allFinite() || !state->_Cov.allFinite() || state->_Cov.diagonal().minCoeff() < 0.)
    dump("invalid posterior covariance", state->_Cov);
